def init_db():
conn = sqlite3.connect("kassa.db")
cursor = conn.cursor()
cursor.execute("""
CREATE TABLE IF NOT EXISTS operatsiyalar (
id INTEGER PRIMARY KEY AUTOINCREMENT,
user_id INTEGER,
user_name TEXT,
turi TEXT,
valyuta TEXT,
summa REAL,
vaqt DATETIME DEFAULT CURRENT_TIMESTAMP
)
""")
cursor.execute("""
CREATE TABLE IF NOT EXISTS xodimlar (
user_id INTEGER PRIMARY KEY,
full_name TEXT,
phone_number TEXT
)
""")
conn.commit()
conn.close()

def db_add_xodim(user_id, full_name, phone_number):
conn = sqlite3.connect("kassa.db")
cursor = conn.cursor()
cursor.execute(
"INSERT OR REPLACE INTO xodimlar (user_id, full_name, phone_number) VALUES (?, ?, ?)",
(user_id, full_name, phone_number)
)
conn.commit()
conn.close()

def db_is_xodim(user_id):
if user_id == DIREKTOR_ID:
return True

cursor.execute("SELECT user_id FROM xodimlar WHERE user_id = ?", (user_id,))

def db_add_operation(user_id, user_name, turi, valyuta, summa):
conn = sqlite3.connect("kassa.db")
cursor = conn.cursor()
cursor.execute(
"INSERT INTO operatsiyalar (user_id, user_name, turi, valyuta, summa) VALUES (?, ?, ?, ?, ?)",
(user_id, user_name, turi, valyuta, summa)
)
conn.commit()
conn.close()

def db_get_balance():
conn = sqlite3.connect("kassa.db")
cursor = conn.cursor()
cursor.execute("SELECT SUM(CASE WHEN turi = 'kirim' THEN summa ELSE -summa END) FROM operatsiyalar WHERE valyuta = 'UZS'")
uzs = cursor.fetchone()[0] or 0.0
cursor.execute("SELECT SUM(CASE WHEN turi = 'kirim' THEN summa ELSE -summa END) FROM operatsiyalar WHERE valyuta = 'USD'")
usd = cursor.fetchone()[0] or 0.0
conn.close()
return uzs, usd

def db_get_staff_report():
conn = sqlite3.connect("kassa.db")
cursor = conn.cursor()
cursor.execute("""
SELECT user_name, valyuta, SUM(summa)
FROM operatsiyalar
WHERE turi = 'kirim'
GROUP BY user_id, valyuta
""")
rows = cursor.fetchall()
conn.close()
return rows
